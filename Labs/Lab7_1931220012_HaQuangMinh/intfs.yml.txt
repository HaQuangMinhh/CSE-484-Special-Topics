- name: Configure routed interfaces on CSR1000v
  hosts: ios
  gather_facts: no

  vars:
    routed_intfs:
      - { name: "GigabitEthernet1", desc: "WAN_UPLINK", ip: "192.168.74.152", mask: "255.255.255.0" }
      - { name: "GigabitEthernet2", desc: "LAN_TRUNK", ip: "10.0.0.1", mask: "255.255.255.0" }

  tasks:
    - name: Apply interface configs
      cisco.ios.ios_config:
        parents: interface {{ item.name }}
        lines:
          - description {{ item.desc }}
          - no shutdown
          - ip address {{ item.ip }} {{ item.mask }}
      loop: "{{ routed_intfs }}"
