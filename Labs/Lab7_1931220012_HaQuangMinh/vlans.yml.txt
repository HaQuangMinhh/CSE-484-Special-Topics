- name: Simulate VLANs using subinterfaces (works 100% on CSR1000v)
  hosts: ios
  gather_facts: no
  vars:
    simulated_vlans:
      - { id: 10, ip: "192.168.10.1", desc: "APP_VLAN" }
      - { id: 20, ip: "192.168.20.1", desc: "DB_VLAN" }
      - { id: 30, ip: "192.168.30.1", desc: "WEB_VLAN" }
  tasks:
    - name: Create subinterface for each VLAN
      cisco.ios.ios_config:
        lines:
          - description "{{ item.desc }}"
          - encapsulation dot1Q {{ item.id }}
          - ip address {{ item.ip }} 255.255.255.0
        parents: interface GigabitEthernet2.{{ item.id }}
      loop: "{{ simulated_vlans }}"
